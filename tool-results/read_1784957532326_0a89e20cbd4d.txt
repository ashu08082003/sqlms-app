   125→function ScoreBadge({
   126→  score,
   127→  failed,
   128→}: {
   129→  score: number
   130→  failed?: number
   131→}) {
   132→  const cls =
   133→    score >= 95
   134→      ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 ring-1 ring-emerald-500/30"
   135→      : score >= 80
   136→        ? "bg-amber-500/15 text-amber-700 dark:text-amber-400 ring-1 ring-amber-500/30"
   137→        : "bg-red-500/15 text-red-700 dark:text-red-400 ring-1 ring-red-500/30"
   138→  return (
   139→    <span
   140→      className={cn(
   141→        "inline-flex items-center rounded-md px-2 py-0.5 text-xs font-semibold",
   142→        cls
   143→      )}
   144→    >
   145→      {score.toFixed(1)}%
   146→      {failed && failed > 0 ? ` · ${failed} fail` : ""}
   147→    </span>
   148→  )
   149→}

   218→function ItemCard({
   219→  idx,
   220→  label,
   221→  state,
   222→  onStatus,
   223→  onReason,
   224→  onPhotoSelect,
   225→  onClearPhoto,
   226→}: {
   227→  idx: number
   228→  label: string
   229→  state: ItemFormState
   230→  onStatus: (s: ItemStatus) => void
   231→  onReason: (r: string) => void
   232→  onPhotoSelect: (f: File | undefined) => void
   233→  onClearPhoto: () => void
   234→}) {
   235→  return (
   236→    <div
   237→      className={cn(
   238→        "rounded-lg border p-3 transition",
   239→        state.status === "NOT_OK" &&
   240→          "border-red-300 dark:border-red-500/40 bg-red-50/40 dark:bg-red-500/5"
   241→      )}
   242→    >
   243→      <div className="flex items-center gap-2">
   244→        <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-md bg-muted text-xs font-semibold text-muted-foreground">
   245→          {idx + 1}
   246→        </span>
   247→        <span className="text-sm font-medium">{label}</span>
   248→      </div>
   249→      <div className="mt-2.5 grid grid-cols-3 gap-2" role="group" aria-label={`Status for ${label}`}>
   250→        {(["OK", "NOT_OK", "NA"] as ItemStatus[]).map((s) => {
   251→          const meta = STATUS_META[s]
   252→          const selected = state.status === s
   253→          return (
   254→            <button
   255→              key={s}
   256→              type="button"
   257→              onClick={() => onStatus(s)}
   258→              aria-pressed={selected}
   259→              aria-label={`${meta.label} for ${label}`}
   260→              className={cn(
   261→                "flex h-11 items-center justify-center rounded-md text-sm font-medium transition focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50",
   262→                selected
   263→                  ? cn(meta.bg, "text-white ring-2", meta.ring)
   264→                  : "border border-input bg-background hover:bg-accent"
   265→              )}
   266→            >
   267→              {meta.label}
   268→            </button>
   269→          )
   270→        })}
   271→      </div>
   272→      <AnimatePresence initial={false}>
   273→        {state.status === "NOT_OK" && (
   274→          <motion.div
   275→            initial={{ height: 0, opacity: 0 }}
   276→            animate={{ height: "auto", opacity: 1 }}
   277→            exit={{ height: 0, opacity: 0 }}
   278→            transition={{ duration: 0.2 }}
   279→            className="overflow-hidden"
   280→          >
   281→            <div className="mt-3 space-y-2">
   282→              <div className="space-y-1.5">
   283→                <Label htmlFor={`reason-${idx}`} className="text-xs text-muted-foreground">
   284→                  Reason (required)
   285→                </Label>
   286→                <Textarea
   287→                  id={`reason-${idx}`}
   288→                  value={state.reason}
   289→                  onChange={(e) => onReason(e.target.value)}
   290→                  placeholder="Describe the issue..."
   291→                  className="min-h-20"
   292→                />
   293→              </div>
   294→              <PhotoInput
   295→                label="Photo (optional, encouraged)"
   296→                photoUrl={state.photoUrl}
   297→                onSelect={onPhotoSelect}
   298→                onClear={onClearPhoto}
   299→              />
   300→            </div>
   301→          </motion.div>
   302→        )}
   303→      </AnimatePresence>
   304→    </div>
   305→  )
   306→}

   308→/* ---------- Main view ---------- */
   309→export function EmployeeScanView({ initialQr }: { initialQr: string | null }) {
   310→  const user = useAppStore((s) => s.user)
   311→  const bump = useAppStore((s) => s.bump)
   312→
   313→  const [phase, setPhase] = useState<Phase>(() =>
   314→    initialQr && extractQr(initialQr) ? "resolving" : "idle"
   315→  )
   316→  const [qrInput, setQrInput] = useState("")
   317→  const [scanning, setScanning] = useState(false)
   318→  const [resolvedCode, setResolvedCode] = useState<string | null>(() =>
   319→    initialQr ? extractQr(initialQr) : null
   320→  )
   321→  const [resolved, setResolved] = useState<ResolveResponse | null>(null)
   322→  const [submitResult, setSubmitResult] = useState<SubmitInspectionResponse | null>(null)
   323→
   324→  const [itemStates, setItemStates] = useState<ItemFormState[]>([])
   325→  const [remarks, setRemarks] = useState("")
   326→  const [overallPhoto, setOverallPhoto] = useState<string | null>(null)
   327→  const [now, setNow] = useState<Date>(() => new Date())
   328→
   329→  /* Recent inspections for the idle screen */
   330→  const recentQuery = useQuery({
   331→    queryKey: ["my-recent-inspections", user?.id],
   332→    queryFn: () =>
   333→      apiFetch<{ inspections: RecentInspection[] }>(
   334→        `/api/inspections?userId=${user!.id}`
   335→      ).then((r) => r.inspections.slice(0, 5)),
   336→    enabled: !!user && phase === "idle",
   337→  })
   338→
   339→  /* Resolve effect — runs whenever we enter the resolving phase. */
   340→  useEffect(() => {
   341→    if (phase !== "resolving" || !resolvedCode) return
   342→    let cancelled = false
   343→    apiFetch<ResolveResponse>(`/api/locations/qr/${resolvedCode}`)
   344→      .then((data) => {
   345→        if (cancelled) return
   346→        if (!data.checklist || data.checklist.items.length === 0) {
   347→          toast.error("No checklist assigned to this location")
   348→          setPhase("idle")
   349→          setResolvedCode(null)
   350→          return
   351→        }
   352→        setResolved(data)
   353→        setItemStates(
   354→          data.checklist.items.map(() => ({
   355→            status: null,
   356→            reason: "",
   357→            photoUrl: null,
   358→          }))
   359→        )
   360→        setRemarks("")
   361→        setOverallPhoto(null)
   362→        setNow(new Date())
   363→        setPhase("form")
   364→      })
   365→      .catch((err) => {
   366→        if (cancelled) return
   367→        toast.error(
   368→          err instanceof Error ? err.message : "QR code not found"
   369→        )
   370→        setPhase("idle")
   371→        setResolvedCode(null)
   372→      })
   373→    return () => {
   374→      cancelled = true
   375→    }
   376→  }, [phase, resolvedCode])
   377→
   378→  function startResolve(code: string) {
   379→    setResolvedCode(code)
   380→    setScanning(false)
   381→    setPhase("resolving")
   382→  }
   383→
   384→  function handleScanResult(decoded: string) {
   385→    const code = extractQr(decoded)
   386→    if (!code) {
   387→      toast.error("Could not read QR code. Try again or enter it manually.")
   388→      return
   389→    }
   390→    startResolve(code)
   391→  }
   392→
   393→  function handleManualSubmit(e: React.FormEvent) {
   394→    e.preventDefault()
   395→    const trimmed = qrInput.trim().toUpperCase()
   396→    if (!/^QR\d+$/.test(trimmed)) {
   397→      toast.error("Enter a valid QR code (e.g. QR000124)")
   398→      return
   399→    }
   400→    setQrInput("")
   401→    startResolve(trimmed)
   402→  }
   403→
   404→  function resetAll() {
   405→    setPhase("idle")
   406→    setResolvedCode(null)
   407→    setResolved(null)
   408→    setSubmitResult(null)
   409→    setItemStates([])
   410→    setRemarks("")
   411→    setOverallPhoto(null)
   412→    setScanning(false)
   413→  }
   414→
   415→  const summary = useMemo(() => {
   416→    let ok = 0
   417→    let notOk = 0
   418→    let na = 0
   419→    for (const s of itemStates) {
   420→      if (s.status === "OK") ok++
   421→      else if (s.status === "NOT_OK") notOk++
   422→      else if (s.status === "NA") na++
   423→    }
   424→    const denom = ok + notOk
   425→    const score = denom === 0 ? 100 : Math.round((ok / denom) * 1000) / 10
   426→    return { ok, notOk, na, score }
   427→  }, [itemStates])
   428→
   429→  const canSubmit = useMemo(() => {
   430→    if (itemStates.length === 0) return false
   431→    for (const s of itemStates) {
   432→      if (!s.status) return false
   433→      if (s.status === "NOT_OK" && !s.reason.trim()) return false
   434→    }
   435→    return true
   436→  }, [itemStates])
   437→
   438→  const isDirty = useMemo(() => {
   439→    if (remarks.trim()) return true
   440→    if (overallPhoto) return true
   441→    return itemStates.some(
   442→      (s) =>
   443→        s.status !== null || s.reason.trim() !== "" || s.photoUrl !== null
   444→    )
   445→  }, [itemStates, remarks, overallPhoto])
   446→
   447→  function setItemStatus(idx: number, status: ItemStatus) {
   448→    setItemStates((prev) =>
   449→      prev.map((s, i) =>
   450→        i === idx ? { ...s, status: s.status === status ? null : status } : s
   451→      )
   452→    )
   453→  }
   454→  function setItemReason(idx: number, reason: string) {
   455→    setItemStates((prev) =>
   456→      prev.map((s, i) => (i === idx ? { ...s, reason } : s))
   457→    )
   458→  }
   459→  function setItemPhoto(idx: number, photoUrl: string | null) {
   460→    setItemStates((prev) =>
   461→      prev.map((s, i) => (i === idx ? { ...s, photoUrl } : s))
   462→    )
   463→  }
   464→
   465→  async function handlePhotoSelect(
   466→    file: File | undefined,
   467→    onSet: (url: string | null) => void
   468→  ) {
   469→    if (!file) return
   470→    if (file.size > MAX_PHOTO_BYTES) {
   471→      toast.warning("Image is too large (max 1.5 MB)")
   472→      return
   473→    }
   474→    try {
   475→      const url = await fileToDataUrl(file)
   476→      onSet(url)
   477→    } catch {
   478→      toast.error("Could not read image file")
   479→    }
   480→  }
   481→
   482→  function handleCancel() {
   483→    if (
   484→      isDirty &&
   485→      !window.confirm("Discard this inspection? Your entries will be lost.")
   486→    )
   487→      return
   488→    resetAll()
   489→  }
   490→
   491→  async function handleSubmit() {
   492→    if (!resolved || !resolved.checklist) return
   493→    if (!canSubmit) {
   494→      for (let i = 0; i < itemStates.length; i++) {
   495→        const s = itemStates[i]
   496→        if (!s.status) {
   497→          toast.error(
   498→            `Please set a status for item ${i + 1}: "${resolved.checklist.items[i]}"`
   499→          )
   500→          return
   501→        }
   502→        if (s.status === "NOT_OK" && !s.reason.trim()) {
   503→          toast.error(
   504→            `Please provide a reason for "${resolved.checklist.items[i]}" (marked Not OK)`
   505→          )
   506→          return
   507→        }
   508→      }
   509→      return
   510→    }
   511→    setPhase("submitting")
   512→    try {
   513→      const responses = itemStates.map((s, i) => ({
   514→        item: resolved.checklist!.items[i],
   515→        status: s.status!,
   516→        reason: s.status === "NOT_OK" ? s.reason.trim() : undefined,
   517→        photoUrl: s.photoUrl ?? undefined,
   518→      }))
   519→      const result = await apiFetch<SubmitInspectionResponse>(
   520→        "/api/inspections",
   521→        {
   522→          method: "POST",
   523→          body: JSON.stringify({
   524→            locationId: resolved.location.id,
   525→            responses,
   526→            remarks: remarks.trim() || undefined,
   527→            photoUrl: overallPhoto ?? undefined,
   528→          }),
   529→        }
   530→      )
   531→      setSubmitResult(result)
   532→      bump()
   533→      setPhase("success")
   534→      toast.success("Inspection submitted")
   535→    } catch (err) {
   536→      toast.error(
   537→        err instanceof Error ? err.message : "Failed to submit inspection"
   538→      )
   539→      setPhase("form")
   540→    }
   541→  }
   542→
   543→  /* ---------- Render ---------- */
   544→  return (
   545→    <AnimatePresence mode="wait">
   546→      <motion.div
   547→        key={phase}
   548→        initial={{ opacity: 0, y: 8 }}
   549→        animate={{ opacity: 1, y: 0 }}
   550→        exit={{ opacity: 0, y: -8 }}
   551→        transition={{ duration: 0.18 }}
   552→      >
   553→        {/* ===== IDLE ===== */}
   554→        {phase === "idle" && (
   555→          <div className="space-y-6">
   556→            <Card className="overflow-hidden border-0 p-0">
   557→              <div className="brand-gradient relative p-6 text-white sm:p-8">
   558→                <div className="absolute -right-10 -top-10 h-32 w-32 rounded-full bg-white/10 blur-2xl" />
   559→                <div className="relative flex items-center gap-3">
   560→                  <div className="rounded-xl bg-white/15 p-2.5 backdrop-blur">
   561→                    <QrCodeIcon className="h-6 w-6" />
   562→                  </div>
   563→                  <div>
   564→                    <h2 className="text-xl font-bold sm:text-2xl">
   565→                      Scan to Inspect
   566→                    </h2>
   567→                    <p className="text-sm text-white/85">
   568→                      Point your camera at the location QR code or enter it manually.
   569→                    </p>
   570→                  </div>
   571→                </div>
   572→              </div>
   573→              <CardContent className="space-y-4 p-4 sm:p-6">
   574→                <Button
   575→                  onClick={() => setScanning((v) => !v)}
   576→                  className="h-12 w-full text-base"
   577→                  variant={scanning ? "outline" : "default"}
   578→                >
   579→                  {scanning ? (
   580→                    <>
   581→                      <X className="mr-2 h-5 w-5" /> Stop camera
   582→                    </>
   583→                  ) : (
   584→                    <>
   585→                      <Camera className="mr-2 h-5 w-5" /> Scan with camera
   586→                    </>
   587→                  )}
   588→                </Button>
   589→
   590→                {scanning && (
   591→                  <QrScanner
   592→                    onScan={handleScanResult}
   593→                    onClose={() => setScanning(false)}
   594→                  />
   595→                )}
   596→
   597→                <div className="relative py-1">
   598→                  <div className="absolute inset-0 flex items-center">
   599→                    <span className="h-px w-full bg-border" />
   600→                  </div>
   601→                  <div className="relative flex justify-center">
   602→                    <span className="bg-card px-3 text-xs uppercase tracking-wide text-muted-foreground">
   603→                      or enter manually
   604→                    </span>
   605→                  </div>
   606→                </div>
   607→
   608→                <form
   609→                  onSubmit={handleManualSubmit}
   610→                  className="flex flex-col gap-2 sm:flex-row"
   611→                >
   612→                  <Input
   613→                    value={qrInput}
   614→                    onChange={(e) => setQrInput(e.target.value)}
   615→                    placeholder="QR000124"
   616→                    className="h-12 text-base font-mono uppercase"
   617→                    autoCapitalize="characters"
   618→                    autoCorrect="off"
   619→                    aria-label="QR code"
   620→                  />
   621→                  <Button
   622→                    type="submit"
   623→                    className="h-12 sm:w-auto"
   624→                    disabled={!qrInput.trim()}
   625→                  >
   626→                    <Keyboard className="mr-2 h-4 w-4" /> Continue
   627→                  </Button>
   628→                </form>
   629→
   630→                {user && (
   631→                  <p className="flex items-center justify-center gap-2 text-xs text-muted-foreground">
